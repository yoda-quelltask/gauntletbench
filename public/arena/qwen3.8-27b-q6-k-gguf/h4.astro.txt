interface Tier {
  name: string;
  priceMonthly: number | null;
  features: string[];
  highlighted?: boolean;
  ctaLabel: string;
  ctaHref: string;
}

interface Props {
  heading: string;
  subheading?: string;
  tiers: Tier[];
}

const {
  heading = "Simple, transparent pricing",
  subheading = "Choose the plan that fits your agency's procurement needs.",
  tiers = [
    {
      name: "Starter",
      priceMonthly: 299,
      features: [
        "Up to 5 active solicitations",
        "Standard compliance checks",
        "Email support"
      ],
      ctaLabel: "Start free trial",
      ctaHref: "/pricing#starter"
    },
    {
      name: "Professional",
      priceMonthly: 799,
      features: [
        "Unlimited solicitations",
        "Advanced compliance automation",
        "Dedicated account manager",
        "API access"
      ],
      highlighted: true,
      ctaLabel: "Start free trial",
      ctaHref: "/pricing#professional"
    },
    {
      name: "Enterprise",
      priceMonthly: null,
      features: [
        "Everything in Professional",
        "Custom integrations",
        "On-premises deployment",
        "24/7 priority support",
        "SLA guarantees"
      ],
      ctaLabel: "Contact sales",
      ctaHref: "/contact"
    }
  ]
} = Astro.props;

const currencyFormatter = new Intl.NumberFormat("en-US", {
  style: "currency",
  currency: "USD",
  minimumFractionDigits: 0,
  maximumFractionDigits: 0
});

function formatPrice(price: number | null): string {
  if (price === null) return "Contact us";
  return currencyFormatter.format(price);
}

<section class="pricing" aria-labelledby="pricing-heading">
  <h2 id="pricing-heading" class="pricing__heading">{heading}</h2>

  {subheading && <p class="pricing__subheading">{subheading}</p>}

  <div class="pricing__grid">
    {tiers.map((tier) => (
      <article
        class:list={[
          "pricing__card",
          { "pricing__card--highlighted": tier.highlighted }
        ]}
      >
        {tier.highlighted && (
          <span class="pricing__badge">Most popular</span>
        )}

        <h3 class="pricing__tier-name">{tier.name}</h3>

        <p class="pricing__price">
          {tier.priceMonthly !== null && (
            <span class="pricing__price-amount">{formatPrice(tier.priceMonthly)}</span>
          )}
          {tier.priceMonthly !== null && (
            <span class="pricing__price-period">/month</span>
          )}
          {tier.priceMonthly === null && (
            <span class="pricing__price-amount pricing__price-amount--contact">Contact us</span>
          )}
        </p>

        <ul class="pricing__features">
          {tier.features.map((feature) => (
            <li class="pricing__feature">{feature}</li>
          ))}
        </ul>

        <a
          class="pricing__cta"
          href={tier.ctaHref}
          aria-label={`${tier.ctaLabel} — ${tier.name} plan`}
        >
          {tier.ctaLabel}
        </a>
      </article>
    ))}
  </div>
</section>
